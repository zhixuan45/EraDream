### 1. 整体结构
剧情文件是一个二进制文件（`.bin`），使用自定义格式存储，包含以下主要部分：

```json
{
  "info": {},          // 剧情基本信息
  "actor_list": {},    // 角色列表
  "bg_list": {},       // 背景列表
  "overlay_list": {},  // 覆盖层列表
  "audio_list": {},    // 音频列表
  "content": []        // 剧情内容序列
}
```

### 2. 各字段详细说明

#### `info` - 剧情信息
```json
{
  "title": "剧情标题",
  "describe": "剧情描述",
  "version": "剧情版本号"  // 可选，默认使用 GameSystem.PLOT_VERSION
}
```

#### `actor_list` - 角色列表
以角色ID为键，每个角色包含：
```json
{
  "display_name": "角色显示名称",
  "list": [
    {
      "actor": "角色资源ID",          // 可选，如果为空则使用 export_actor
      "portrait_index": 0,           // 立绘索引
      "export_actor": Image,         // 导出的立绘图像（二进制）
      "sticker_offset": [0, 0],      // 贴纸偏移
      "sticker_scale": 1.0           // 贴纸缩放
    }
  ]
}
```

#### `bg_list` - 背景列表
以背景ID为键，每个背景包含：
```json
{
  "background": "背景图片路径",       // 可选
  "colorful_background": false,      // 是否为纯色背景
  "background_color": "#FFFFFF",     // 纯色背景的颜色
  "export_image": Image,             // 导出的背景图像（二进制）
  "export_video": PackedByteArray,   // 导出的视频数据（二进制）
  "video_start_time": 0.0,           // 视频开始时间
  "video_end_time": 0.0,             // 视频结束时间
  "video_loop": false,               // 是否循环播放
  "video_audio": false,              // 是否播放视频音频
  "default": false                   // 是否为默认背景
}
```

#### `overlay_list` - 覆盖层列表
以覆盖层ID为键，每个覆盖层包含：
```json
{
  "display_name": "覆盖层显示名称",
  "overlay_path": "覆盖层图片路径",   // 可选
  "export_image": Image              // 导出的覆盖层图像（二进制）
}
```

#### `audio_list` - 音频列表
以音频ID为键，每个音频包含：
```json
{
  "audio_path": "音频文件路径",       // 可选
  "export_audio": AudioStream,       // 导出的音频数据（二进制）
  "default": false,                  // 是否为默认音频
  "default_loop": false,             // 默认是否循环播放
  "global_start_time": 0.0,          // 全局开始时间
  "global_end_time": 0.0             // 全局结束时间
}
```

#### `content` - 剧情内容序列
这是一个数组，包含多种类型的剧情指令：

### 3. 剧情指令类型

#### 1) `dialog` - 对话
```json
{
  "type": "dialog",
  "actor": "角色ID",
  "portrait": {
    "index": 0,                      // 立绘索引
    "position": [x, y],              // 位置
    "old_position": [x, y],          // 旧位置（用于动画）
    "scale": 1.0,                    // 缩放
    "module": "ffffff",              // 颜色（十六进制）
    "old_module": "ffffff",          // 旧颜色
    "module_anim": false,            // 是否颜色动画
    "scale_anim": false,             // 是否缩放动画
    "animation": 0,                  // 动画类型（枚举值）
    "sticker": "",                   // 贴纸类型
    "hide": false,                   // 是否隐藏
    "exit_after_dialogue": false,    // 对话后退出
    "hide_dialogue_after_exit": false, // 退出后隐藏对话
    "light": true,                   // 是否高亮
    "clear_other_highlight": true    // 清除其他高亮
  },
  "temp_display_name": "临时显示名称", // 可选
  "content": [
    "文本内容",                      // 简单文本
    {
      "text": "文本内容",
      "voice": AudioStream           // 语音音频
    }
  ],
  "auto_skip": 2,                    // 自动跳过时间（秒）
  "wait_anim": false                 // 等待动画完成
}
```

#### 2) `speak` - 旁白
```json
{
  "type": "speak",
  "content": [
    "文本内容",
    {
      "text": "文本内容",
      "voice": AudioStream
    }
  ],
  "auto_skip": 2                     // 自动跳过时间（秒）
}
```

#### 3) `portrait` - 立绘操作
```json
{
  "type": "portrait",
  "actor": "角色ID",
  "portrait": {
    // 同 dialog.portrait 结构
    "index": 0,
    "position": [x, y],
    "old_position": [x, y],
    "scale": 1.0,
    "module": "ffffff",
    "old_module": "ffffff",
    "module_anim": false,
    "scale_anim": false,
    "animation": 0,
    "sticker": "",
    "hide": false,
    "light": false,
    "clear_other_highlight": true
  },
  "wait": 0.0,                       // 等待时间（秒）
  "skip_immediately": false,         // 是否立即跳过
  "wait_anim": false                 // 等待动画完成
}
```

#### 4) `background` - 背景切换
```json
{
  "type": "background",
  "background": "背景ID",
  "change_type": 0,                  // 切换类型（枚举值）
  "filter": 0,                       // 滤镜类型（枚举值）
  "hide": false,                     // 是否隐藏界面
  "wait": 0.0,                       // 等待时间（秒）
  "skip_immediately": false          // 是否立即跳过
}
```

#### 5) `title` - 标题显示
```json
{
  "type": "title",
  "content": "标题文本",
  "big_title": "大标题文本"           // 可选
}
```

#### 6) `choice` - 选项分支
```json
{
  "type": "choice",
  "content": [
    {
      "name": "选项名称",
      "content": []                  // 分支剧情内容
    }
  ]
}
```

#### 7) `shake_window` - 窗口震动
```json
{
  "type": "shake_window",
  "time": 1.0,                       // 震动时间（秒）
  "wait": 0.0,                       // 等待时间（秒）
  "skip_immediately": false          // 是否立即跳过
}
```

#### 8) `overlay` - 覆盖层显示
```json
{
  "type": "overlay",
  "overlay": "覆盖层ID",
  "position": [x, y],                // 位置
  "scale": 1.0,                      // 缩放
  "time": 2.0,                       // 显示时间（秒）
  "hide": false,                     // 是否隐藏
  "darken_background": true,         // 是否暗化背景
  "auto_time": true,                 // 是否自动计时
  "skip_immediately": false          // 是否立即跳过
}
```

#### 9) `audio` - 音频控制
```json
{
  "type": "audio",
  "audio": "音频ID",
  "is_bgm": false,                   // 是否为背景音乐
  "is_stop": false,                  // 是否停止
  "start_time": 0.0,                 // 开始时间（-1表示使用全局时间）
  "end_time": 0.0,                   // 结束时间（-1表示使用全局时间）
  "during_time": 1.0,                // 过渡时间（秒）
  "is_loop": false,                  // 是否循环
  "time": 0.0,                       // 等待时间（秒）
  "skip_immediately": false          // 是否立即跳过
}
```

#### 10) `status` - 状态显示
```json
{
  "type": "status",
  "path": "状态图片路径",
  "time": 2.0                        // 显示时间（秒）
}
```

#### 11) `feature` - 特殊功能
```json
{
  "type": "feature",
  "features": [0, 1],                // 功能ID数组（枚举值）
  "wait_time": 0.5,                  // 等待时间（秒）
  "skip_immediately": false          // 是否立即跳过
}
```

### 5. 文本格式说明

#### 暂停标签
文本中可以包含暂停标签，格式为：`{pause:毫秒数}`
例如：`"你好{pause:500}世界"` 会在"你好"后暂停500毫秒

#### BBCode支持
文本通过 `GameSystem.text_style_helper.convert_to_bbcode()` 转换为BBCode格式，支持富文本样式。

### 6. 二进制数据格式

剧情文件使用自定义二进制格式：
```json
{
  "__uma_bin__": true,               // 标识符
  "compressed": true,                // 是否压缩
  "size": 12345,                     // 原始数据大小
  "data": PackedByteArray            // 数据（可能压缩）
}
```

压缩使用DEFLATE算法，可以通过设置 `editor/disable_project_file_compression` 禁用压缩。

### 7. 资源序列化

资源（图像、音频）通过 `GameSystem.data_helper.serialize_resources_in_data()` 序列化为二进制数据，反序列化使用 `deserialize_resources_in_data()`。

### 8. 坐标系统

位置坐标使用原始分辨率（`default_resolution`）下的坐标，在播放时通过 `get_scaled_position()` 函数根据当前窗口大小进行缩放。
